   3 // RER Platform (RER Managed) applies migrations/core itself and records each

  50 // Resolve ledger identities against the destination's own migration files.
  51 // Hash-bearing entries must identify exactly one file; a coinciding name may
  52 // confirm that file, but cannot point at another one. Hashless rows use exact
  53 // names or seed-declared aliases. The caller keeps the destination replay order.
  54 export function matchMigrationLedger(entries,files,{dataOnly=[]}={}) {
  55   const matched=entries.flatMap(m=>{
  56     const named=files.filter(file=>file.name===m.name || file.aliases?.includes(m.name));
  57     insist(named.length<=1,'migration_ledger:ambiguous:'+m.name,409);
  58     if(m.sha256===undefined) {
  59       if(!named.length && dataOnly.includes(m.name))return [];
  60       insist(named.length,'migration_ledger:unknown:'+m.name,409);
  61       return [named[0].name];
  62     }
  63     const matches=files.filter(file=>file.sha256===m.sha256);
  64     if(!matches.length && !named.length && dataOnly.includes(m.name))return [];
  65     insist(matches.length,'migration_ledger:unknown:'+m.name,409);
  66     insist(matches.length===1,'migration_ledger:ambiguous:'+m.name,409);
  67     insist(!named.length || named[0]===matches[0],'migration_ledger:conflict:'+m.name,409);
  68     return [matches[0].name];
  69   });
  70   insist(new Set(matched).size===matched.length,'migration_ledger:ambiguous',409);
  71   return matched;
  72 }
